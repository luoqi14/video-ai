server {
    server_name tools-video.lumiidental.com;

    # 增加上传大小和超时限制
    client_max_body_size 500M;
    proxy_connect_timeout 300s;
    proxy_send_timeout 600s;
    proxy_read_timeout 24h;

    # --- 跨源隔离 Header ---
    add_header Cross-Origin-Opener-Policy "same-origin" always;
    add_header Cross-Origin-Embedder-Policy "require-corp" always;

    # --- 静态资源处理 (新增) ---
    # 处理 Next.js 静态资源
    location /_next/static/ {
        add_header Cross-Origin-Opener-Policy "same-origin" always;
        add_header Cross-Origin-Embedder-Policy "require-corp" always;
        proxy_pass http://127.0.0.1:3002;
        proxy_set_header Host $host;
        proxy_cache_valid 200 1y;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # 处理 favicon 和其他根目录静态文件
    location ~* \.(ico|png|jpg|jpeg|gif|svg|woff|woff2|ttf|eot)$ {
        add_header Cross-Origin-Opener-Policy "same-origin" always;
        add_header Cross-Origin-Embedder-Policy "require-corp" always;
        proxy_pass http://127.0.0.1:3002;
        proxy_set_header Host $host;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # --- 缓存策略优化 ---
    location ~* \.(js|css|wasm)$ {
        add_header Cross-Origin-Opener-Policy "same-origin" always;
        add_header Cross-Origin-Embedder-Policy "require-corp" always;
        proxy_pass http://127.0.0.1:3002;
        proxy_set_header Host $host;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }

    # 将 /api/ 开头的请求转发给后端服务
    location /api/ {
        proxy_buffering off;
        gzip off;
        proxy_pass http://127.0.0.1:8002;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 将所有其他请求转发给前端服务
    location / {
        proxy_pass http://127.0.0.1:3002;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    listen [::]:443 ssl http2;
    listen 443 ssl http2;
    ssl_certificate /etc/letsencrypt/live/dify.lumiidental.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/dify.lumiidental.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

server {
    if ($host = tools-video.lumiidental.com) {
        return 301 https://$host$request_uri;
    }
    listen 80;
    listen [::]:80;
    server_name tools-video.lumiidental.com;
    return 404;
}
